        // ============================================================
        // ===== АНИМИРОВАННЫЙ ФОН (человек на горе, трава, бабочка) =====

        const canvas = document.getElementById('bgCanvas');
        const ctx = canvas.getContext('2d');

        let W, H;

        function resize() {
            W = canvas.width = window.innerWidth;
            H = canvas.height = window.innerHeight;
        }
        window.addEventListener('resize', resize);
        resize();

        // ---- ВРЕМЯ ----
        let time = 0;

        // ---- ТРАВА ----
        const grassBlades = [];
        for (let i = 0; i < 200; i++) {
            grassBlades.push({
                x: Math.random() * W,
                y: H * 0.7 + Math.random() * H * 0.3,
                height: Math.random() * 25 + 15,
                width: Math.random() * 2 + 1,
                phase: Math.random() * Math.PI * 2,
                speed: Math.random() * 0.02 + 0.01,
                bend: Math.random() * 0.3 + 0.1,
                color: `hsl(${120 + Math.random() * 30}, ${40 + Math.random() * 30}%, ${20 + Math.random() * 15}%)`
            });
        }

        // ---- БАБОЧКА ----
        let butterfly = {
            x: 0,
            y: 0,
            targetX: 0,
            targetY: 0,
            wingAngle: 0,
            phase: 0,
            speed: 0.008,
            sitting: false,
            sitTimer: 0
        };

        // ---- ЛУЧИ СОЛНЦА ----
        const rays = [];
        for (let i = 0; i < 12; i++) {
            rays.push({
                angle: (i / 12) * Math.PI * 2 + Math.random() * 0.2,
                length: Math.random() * 0.3 + 0.5,
                width: Math.random() * 4 + 2,
                speed: Math.random() * 0.005 + 0.002
            });
        }

        // ---- РИСОВАНИЕ СЦЕНЫ ----
        function drawScene(timestamp) {
            time = timestamp;

            // Очистка
            ctx.clearRect(0, 0, W, H);

            // ----- НЕБО (градиент) -----
            const skyGrad = ctx.createLinearGradient(0, 0, 0, H * 0.7);
            skyGrad.addColorStop(0, '#0a0010');
            skyGrad.addColorStop(0.3, '#1a0a2e');
            skyGrad.addColorStop(0.6, '#2a1a3e');
            skyGrad.addColorStop(0.8, '#4a2a4e');
            skyGrad.addColorStop(1, '#2a1a3e');
            ctx.fillStyle = skyGrad;
            ctx.fillRect(0, 0, W, H);

            // ----- СОЛНЦЕ (свет) -----
            const sunX = W * 0.3;
            const sunY = H * 0.1;

            // Свечение солнца
            const sunGlow = ctx.createRadialGradient(sunX, sunY, 0, sunX, sunY, Math.min(W, H) * 0.6);
            sunGlow.addColorStop(0, 'rgba(255, 200, 150, 0.15)');
            sunGlow.addColorStop(0.3, 'rgba(255, 180, 130, 0.08)');
            sunGlow.addColorStop(0.7, 'rgba(200, 150, 100, 0.03)');
            sunGlow.addColorStop(1, 'rgba(100, 50, 30, 0)');
            ctx.fillStyle = sunGlow;
            ctx.beginPath();
            ctx.arc(sunX, sunY, Math.min(W, H) * 0.6, 0, Math.PI * 2);
            ctx.fill();

            // Само солнце
            const sunGrad = ctx.createRadialGradient(sunX, sunY, 0, sunX, sunY, Math.min(W, H) * 0.08);
            sunGrad.addColorStop(0, 'rgba(255, 220, 180, 0.6)');
            sunGrad.addColorStop(0.5, 'rgba(255, 200, 150, 0.3)');
            sunGrad.addColorStop(1, 'rgba(200, 150, 100, 0)');
            ctx.fillStyle = sunGrad;
            ctx.beginPath();
            ctx.arc(sunX, sunY, Math.min(W, H) * 0.08, 0, Math.PI * 2);
            ctx.fill();

            // ----- ЛУЧИ СОЛНЦА -----
            rays.forEach((ray, i) => {
                const angle = ray.angle + time * ray.speed;
                const length = ray.length * Math.min(W, H) * 0.5;
                const startX = sunX + Math.cos(angle) * 20;
                const startY = sunY + Math.sin(angle) * 20;
                const endX = sunX + Math.cos(angle) * length;
                const endY = sunY + Math.sin(angle) * length;

                ctx.save();
                ctx.globalAlpha = 0.02 + 0.02 * (0.5 + 0.5 * Math.sin(time * 0.001 + i));
                ctx.strokeStyle = 'rgba(255, 220, 180, 0.3)';
                ctx.lineWidth = ray.width * (0.5 + 0.5 * Math.sin(time * 0.002 + i));
                ctx.beginPath();
                ctx.moveTo(startX, startY);
                ctx.lineTo(endX, endY);
                ctx.stroke();
                ctx.restore();
            });

            // ----- ГОРЫ -----
            // Дальние горы
            ctx.fillStyle = 'rgba(30, 15, 40, 0.5)';
            ctx.beginPath();
            ctx.moveTo(0, H * 0.6);
            const peaks1 = [
                [0, 0.6], [0.05, 0.55], [0.12, 0.58], [0.18, 0.52], [0.25, 0.56],

                [0.65, 0.52], [0.72, 0.46], [0.78, 0.50], [0.85, 0.44], [0.92, 0.49], [1, 0.45]
            ];

            ctx.lineTo(0, H);

                    butterfly.sitting = true;
                    butterfly.sitTimer = 0;
                    butterfly.targetX = personX + (Math.random() - 0.5) * scale * 0.3;

                if (butterfly.sitTimer > 5 + Math.random() * 3) {
                    butterfly.sitting = false;
                    butterfly.phase = Math.random() * Math.PI * 2;

        let captchaAnswer = 0;

        function generateCaptcha() {
            const num1 = Math.floor(Math.random() * 15) + 1;
            const num2 = Math.floor(Math.random() * 15) + 1;
            const operators = ['+', '-'];
            const op = operators[Math.floor(Math.random() * operators.length)];

            let display, answer;

            if (op === '+') {
                display = `${num1} + ${num2}`;
                answer = num1 + num2;
            } else {
                if (num1 >= num2) {
                    display = `${num1} - ${num2}`;
                    answer = num1 - num2;
                } else {
                    display = `${num2} - ${num1}`;
                    answer = num2 - num1;
                }
            }

            document.getElementById('captchaDisplay').textContent = display;
            captchaAnswer = answer;
            document.getElementById('captchaInput').value = '';
        }
